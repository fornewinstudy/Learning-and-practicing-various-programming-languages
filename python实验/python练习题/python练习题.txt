def is_prime(n):
    if n>1:
        for item in range(2,n):
            if n%item==0:
                break
        else:
            return True
        return False
    else:

def l1_distance(x, y):# 曼哈顿距离
    n = len(x)
    distance = 0
    for i in range(n):
        distance += abs(x[i] - y[i])
    return distance

import math
def l2_distance(x, y):# 欧式距离
    n = len(x)
    distance = 0
    for i in range(n):
        distance += (x[i] - y[i]) ** 2
    return math.sqrt(distance)
print('x=(3,4,1),y=(1,5,2)') # 三维向量
print('l1:', l1_distance((3,4,1),(1,5,2)))
print('l2:', l2_distance((3,4,1),(1,5,2)))

def find_key_by_values(dct, value): # 找到所有的关联键
    # 使用字典推导式查找关联键
    keys = [key for key, val in dct.items() if val == value]
    # 找到关联键则返回
    if keys:
        return keys
    # 如果没有找到关联键，则返回 None
    else:
        return None

def find_key_by_value(dic, a): # 返回第一个关联键
    for item in dic:
        if dic[item] == a:
            return item
            break
    else:
        return None

dic1 = {'张三': 1, '王五': 2, '李四': 3, 'a': 3, "B": 4}
print('value:', find_key_by_value(dic1, 3))
print('numerical list(values):', find_key_by_values(dic1, 3))

# 密集向量相加
def dense_vector_add(v1, v2): # 都是列表
    if len(v1) != len(v2):
        raise ValueError("Vectors must have the same length")
    return [v1[i] + v2[i] for i in range(len(v1))]
# (密集)向量相乘(即内积)
def dense_vector_mult(v1, v2):
    if len(v1) != len(v2):
        raise ValueError("Vectors must have the same length")
    return sum([v1[i] * v2[i] for i in range(len(v1))])
# 稀疏向量相加
def sparse_vector_add(s1, s2): # 都是字典
    values1 = list(s1.values()) # 值视图
    keys1 = list(s1.keys()) # 键视图
    values2 = list(s2.values())
    keys2 = list(s2.keys())
    # 字典对应向量（列表）的初始化
    v1 = [i*0 for i in range(keys1[-1]+1)]
    v2 = [i*0 for i in range(keys2[-1]+1)]
    for item in range(keys1[-1]+1): # 锁定向量长度
        if item == keys1[0] : # 找到非零值
            v1[item] = values1[0] # 幅值非零值
            keys1.pop(0) # 更新键视图
            values1.pop(0) # 更新值视图
        else:
            v1[item] = 0 # 其他的元素都为0
    # 同上
    for item in range(keys2[-1]+1):
        if item == keys2[0]:
            v2[item] = values2[0]
            keys2.pop(0)
            values2.pop(0)
        else:
            v2[item] = 0
    if len(v1) != len(v2):
        raise ValueError("Vectors must have the same length")
    return [v1[i] + v2[i] for i in range(len(v1))]
# 稀疏向量相乘
def sparse_vector_multiply(s1, s2):
    # 同sparse_vector_add内一致
    values1 = list(s1.values())
    keys1 = list(s1.keys())
    values2 = list(s2.values())
    keys2 = list(s2.keys())
    v1 = [i * 0 for i in range(keys1[-1] + 1)]
    v2 = [i * 0 for i in range(keys2[-1] + 1)]
    for item in range(keys1[-1] + 1):
        if item == keys1[0]:
            v1[item] = values1[0]
            keys1.pop(0)
            values1.pop(0)
        else:
            v1[item] = 0
    for item in range(keys2[-1] + 1):
        if item == keys2[0]:
            v2[item] = values2[0]
            keys2.pop(0)
            values2.pop(0)
        else:
            v2[item] = 0
    if len(v1) != len(v2):
        raise ValueError("Vectors must have the same length")
    return sum([v1[i] * v2[i] for i in range(len(v1))])

def root(f, a, b): # 求root函数
    c=0
    if f(a) == 0:
        return a
    elif f(b) == 0:
        return b
    elif f(a) * f(b) > 0: # 判断是否为相同符号
        # raise ValueError("函数值具有相同符号")
        return "函数值具有相同符号"
    if b < a: # 转换标准大小关系
        a, b=b, a
    while abs(f(c)-0) > 1e-7: # 设定求根截止数:1e-6
        c = (a + b) / 2  # 二分法求值
        if f(c) == 0:
            return c
        elif f(c) < 0:
            a = c
        else:
            b = c
    return (a + b) / 2

def root2(f,a,b): # 函递归求root数
    if f(a) == 0:
        return a
    elif f(b) == 0:
        return b
    elif f(a) * f(b) > 0: # 判断是否为相同符号
        return "函数值具有相同符号"
    if b < a:
        a, b = b, a # 转换标准大小关系
    c = (a + b) / 2 # 二分法求值
    if f(c) == 0 or abs(f(c)-0) < 1e-7: # 截止条件：c等于0或偏差小于1e-6
        return c
    elif f(c) < 0:
        a = c
    else:
        b = c
    return root2(f, a, b) # 递归
